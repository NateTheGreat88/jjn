// Firestore Security Rules for JJN Chat
// Paste these in the Firestore "Rules" tab in Firebase Console

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow anyone to read and write messages
    // For production, you might want to add authentication
    match /messages/{messageId} {
      allow read, write: if true;
    }
    
    // Allow anyone to read and write room messages (private rooms)
    match /roomMessages/{messageId} {
      allow read, write: if true;
    }
    
    // Allow anyone to read and write room metadata
    match /rooms/{roomId} {
      allow read, write: if true;
    }
    
    // User data - users can read/write their own data
    match /users/{userId} {
      // Users can read/write their own data
      allow read, write: if request.auth != null && request.auth.uid == userId;
      // Authenticated users can read any user (needed for dev operations like granting items)
      allow read: if request.auth != null;
      // Authenticated users can update any user (devs need this to grant items to other users)
      // Note: The UI already restricts grant functionality to devs via isDev() check
      allow update: if request.auth != null;
      // Only allow users to create their own document
      allow create: if request.auth != null && request.auth.uid == userId;
    }
    
    // Purchases - allow anyone to create purchases, devs can read, update, and delete
    match /purchases/{purchaseId} {
      allow create: if true;
      allow read: if request.auth != null; // Authenticated users (devs) can read
      allow update: if request.auth != null; // Authenticated users (devs) can update to grant items
      allow delete: if request.auth != null; // Authenticated users (devs) can delete for cleanup
    }
    
    // CD Orders - allow anyone to create, read recent orders (for public display), and update if admin
    match /cd_orders/{orderId} {
      allow create: if true;
      allow read: if true; // Public display of recent orders
      allow update: if request.auth != null; // Authenticated users (admins) can update (mark shipped)
      allow delete: if false; // No deletion allowed
    }
    
    // CD Reviews - allow anyone to create and read, users can delete their own, admins can delete any
    match /cd_reviews/{reviewId} {
      allow create: if true;
      allow read: if true; // Public display of reviews
      allow update: if false; // No updates allowed
      allow delete: if request.auth != null && (
        resource.data.userId == request.auth.uid || // User can delete their own review
        true // Admins can delete any (you may want to add admin check here)
      );
    }
    
  }
}
